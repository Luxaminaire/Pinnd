CLASS Pinn
    // Constructor initializes a new note
    CONSTRUCTOR(title, content)
        SET this.title = title
        SET this.content = content
    END CONSTRUCTOR
END CLASS


CLASS PinnBoard
    // Constructor initializes an empty board
    CONSTRUCTOR()
        SET this.pins = EMPTY LIST
    END CONSTRUCTOR

    // Method to add a new pin to the board
    METHOD add_pinn(new_pinn)
        APPEND new_pinn TO this.pins
    END METHOD

    // Method to search for a keyword across all pins
    METHOD search(search_query)
        
        // Handle empty search strings early
        IF search_query IS EMPTY THEN
            RETURN EMPTY LIST
        END IF

        // Convert the search query to lowercase for case-insensitive matching
        SET search_term = LOWERCASE(search_query)
        SET search_results = EMPTY LIST

        // Loop through every pin currently on the board
        FOR EACH pinn IN this.pins
            
            // Convert both the title and content to lowercase to check against the search term
            SET title_lower = LOWERCASE(pinn.title)
            SET content_lower = LOWERCASE(pinn.content)

            // Check if the search term exists in either the title or the content
            IF search_term IS IN title_lower OR search_term IS IN content_lower THEN
                APPEND pinn TO search_results
            END IF
            
        END FOR

        // Return the final list of matching pins
        RETURN search_results
        
    END METHOD
END CLASS
